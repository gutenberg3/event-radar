    async function setRSVP(eventId, status) {

      if (!userInitials) {

        initialsInput.focus();

        initialsInput.classList.add('ring-2', 'ring-[#00ffcc]');

        setTimeout(() => initialsInput.classList.remove('ring-2', 'ring-[#00ffcc]'), 1500);

        return;

      }


      const ev = eventsData.find(e => e.id === eventId);

      if (!ev) return;


      let y = (ev.yAttendees || '').split(';').filter(Boolean);

      let m = (ev.mAttendees || '').split(';').filter(Boolean);

      let n = (ev.nAttendees || '').split(';').filter(Boolean);


      const currentStatus = y.includes(userInitials) ? 'Y' : m.includes(userInitials) ? 'M' : n.includes(userInitials) ? 'N' : null;


      y = y.filter(i => i !== userInitials);

      m = m.filter(i => i !== userInitials);

      n = n.filter(i => i !== userInitials);


      if (currentStatus !== status) {

        if (status === 'Y') y.push(userInitials);

        if (status === 'M') m.push(userInitials);

        if (status === 'N') n.push(userInitials);

      }


      ev.yAttendees = y.join(';');

      ev.mAttendees = m.join(';');

    function autoFetchMetadata() {

      const urlInput = document.getElementById('modal-url');

      const val = urlInput.value.trim();

      if (!val) return;


      const statusEl = document.getElementById('scan-status');

      statusEl.classList.remove('hidden');

      statusEl.textContent = '> Scanning URL...';


      try {

        const u = new URL(val);

        const host = u.hostname.replace('www.', '');

        const slug = u.pathname.replace(/^\/|\/$/g, '').split('/').pop() || '';

        

        let derivedTitle = host.split('.')[0].toUpperCase();

        if (slug && !slug.includes('event') && !slug.includes('websitePage')) {

          derivedTitle += ' ' + slug.replace(/[-_]/g, ' ').toUpperCase();

        }


        const titleEl = document.getElementById('modal-title');

        if (!titleEl.value) titleEl.value = derivedTitle;


        statusEl.textContent = '> Extracted from URL';

      } catch (e) {

        statusEl.textContent = '> Manual entry mode';

      }

    }

    function saveEvent() {

      const eventId = document.getElementById('modal-event-id').value;

      const url = document.getElementById('modal-url').value.trim();

      if (!url) return;


      let title = document.getElementById('modal-title').value.trim();

      let dateStr = document.getElementById('modal-date-str').value.trim();

      let startDate = document.getElementById('modal-start-date').value;

      let location = document.getElementById('modal-location').value.trim();

      let sponsors = document.getElementById('modal-sponsors').value.trim();


      if (!title) {

        try { title = new URL(url).hostname.replace('www.', ''); } catch { title = "New Event"; }

      }

      if (!startDate) startDate = "2026-12-31";

      if (!dateStr) dateStr = startDate;


      if (eventId) {

        // EDIT EXISTING

        const ev = eventsData.find(e => e.id === eventId);

        if (ev) {

          ev.url = url;

          ev.title = title;

          ev.dateStr = dateStr;

          ev.startDate = startDate;

          ev.location = location;

          ev.sponsors = sponsors;


          renderEvents();

          closeModal();


          try {

            fetch(SHEET_API_URL, {

              method: 'POST',

              mode: 'no-cors',

              headers: { 'Content-Type': 'text/plain' },

              body: JSON.stringify({

                action: 'edit',

                id: ev.id,

                title,

                startDate,

                dateStr,

                location,

                url,

                sponsors

              })

            });

          } catch (err) {

            console.error('Failed to sync edit', err);

          }

        }

      } else {

        // ADD NEW

        const newEv = {

          id: 'ev-' + Date.now(),

          title,

          startDate,

          endDate: '',

          dateStr,

          location,

          url,

          sponsors,

          yAttendees: userInitials || '',

          mAttendees: '',

          nAttendees: ''

        };


        eventsData.push(newEv);

        renderEvents();

        closeModal();


        try {

          fetch(SHEET_API_URL, {

            method: 'POST',

            mode: 'no-cors',

            headers: { 'Content-Type': 'text/plain' },

            body: JSON.stringify({ action: 'add', ...newEv })

          });

        } catch (err) {

          console.error('Failed to append to Sheet', err);

        }

      }

    }


    async function sendRelayMessage() {

      const msgInput = document.getElementById('email-message-body');

      const contactInput = document.getElementById('email-sender-contact');

      const statusEl = document.getElementById('email-relay-status');

      const btn = document.getElementById('btn-send-relay');

      

      const message = msgInput.value.trim();

      btn.disabled = true;

      btn.textContent = 'TRANSMITTING...';

      statusEl.classList.remove('hidden');

      statusEl.textContent = 'Sending...';


      try {

        await fetch(SHEET_API_URL, {

          method: 'POST',

          mode: 'no-cors',

          headers: { 'Content-Type': 'text/plain' },

          body: JSON.stringify({

            action: 'relay_email',

            senderId: userInitials,

            senderContact: contactInput.value.trim(),

            message: message

          })

        });


        statusEl.textContent = '✓ Message transmitted to Cobb.';

        statusEl.className = 'text-[10px] text-center text-[#00ffcc] font-mono';

        msgInput.value = '';

        setTimeout(() => { closeWhoDis(); }, 1600);

      } catch (e) {

        statusEl.textContent = 'Transmission failed. Try again.';

        statusEl.className = 'text-[10px] text-center text-red-400 font-mono';

      } finally {

        btn.disabled = false;

        btn.textContent = 'TRANSMIT MESSAGE';

      }

    }
